def add_curated_topscorers(merged_players):
    by_name = {}
    for player in merged_players.values():
        keys = {norm(player.get("name", ""))}
        keys.update(norm(alias) for alias in (player.get("aliases") or []))
        for key in keys:
            if key:
                by_name.setdefault(key, player)

    added = 0
    missing = []

    for award, names in TOPSCORER_AWARDS.items():
        for name in names:
            player = by_name.get(norm(name))
            if not player:
                missing.append((award, name))
                continue

            player.setdefault("awards", [])
            if not any(norm(value) == norm(award) for value in player["awards"]):
                player["awards"].append(award)
                added += 1

    return added, missing

JUNK = (
    "career break","inactive","retired","retirement","without club","no club",
    "free agent","unknown","end of career","not active","reserves","reserve",
    "youth"," u21"," u20"," u19"," u18"," u17"," u16"," u15"
)

def norm(value: str) -> str:
    value = str(value or "").strip()
    value = unicodedata.normalize("NFKD", value)
    value = "".join(ch for ch in value if not unicodedata.combining(ch))
    value = value.casefold()
    value = re.sub(r"\s+", " ", value)
    return value

def clean_name(value: str) -> str:
    return re.sub(r"\s*\(\d+\)\s*$", "", str(value or "")).strip()

def canonical(value: str) -> str:
    raw = str(value or "").strip()
    return CANONICAL.get(norm(raw), raw)

def canonical_country(value: str) -> str:
    raw = str(value or "").strip()
    return COUNTRY_CANONICAL.get(norm(raw), canonical(raw))

def split_countries(values):
    """Transfermarkt zet dubbele nationaliteiten in één tekst met dubbele spaties."""
    out = []
    for value in values or []:
        parts = re.split(r"\s{2,}", str(value or "").strip())
        out.extend(part for part in parts if part)
    return out

def unique(values, field="generic"):
    out, seen = [], set()
    for value in values or []:
        value = canonical_country(value) if field == "country" else canonical(value)
        key = norm(value)
        if value and key not in seen:
            seen.add(key)
            out.append(value)
    return out

def is_junk(value: str) -> bool:
    n = f" {norm(value)} "
    return any(term in n for term in JUNK)

# Eerst samenvoegen op stabiel bron-ID. Alleen records zonder ID vallen terug op naam.
for raw in records:
    name = clean_name(raw.get("name", ""))
    if not name:
        continue

    source_id = str(raw.get("id") or "").strip()
    name_keys = [norm(name)]
    name_keys.extend(
        norm(clean_name(alias))
        for alias in (raw.get("aliases") or [])
        if clean_name(alias)
    )
    existing_key = next((name_to_key[n] for n in name_keys if n in name_to_key), None)
    key = f"id:{source_id}" if source_id else (existing_key or f"name:{norm(name)}")
    player = merged.setdefault(key, {
        "id": source_id or f"NAME-{norm(name).replace(' ', '-')}",
        "name": name,
        "aliases": [],
        "clubs": [],
        "internationalTeams": [],
        "awards": [],
        "competitions": [],
        "image": raw.get("image") or raw.get("imageUrl") or raw.get("photo") or None,
        "legend": bool(raw.get("legend"))
    })

    player["aliases"].extend(raw.get("aliases") or [])
    player["clubs"].extend(raw.get("clubs") or [])
    player["internationalTeams"].extend(split_countries(raw.get("internationalTeams") or []))
    player["awards"].extend(raw.get("awards") or [])
    player["competitions"].extend(raw.get("competitions") or [])
    if not player["image"]:
        player["image"] = raw.get("image") or raw.get("imageUrl") or raw.get("photo") or None
    player["legend"] = player["legend"] or bool(raw.get("legend"))
    for name_key in name_keys:
        if name_key:
            name_to_key.setdefault(name_key, key)

topscorer_added, topscorer_missing = add_curated_topscorers(merged)

# Daarna alleen spelrelevante waarden bewaren.

    player["aliases"] = unique([clean_name(x) for x in player["aliases"] if clean_name(x)])
    player["clubs"] = unique([x for x in player["clubs"] if not is_junk(x)])
    player["internationalTeams"] = unique(
        [x for x in player["internationalTeams"] if not is_junk(x)],
        field="country"
    )
    player["awards"] = unique([x for x in player["awards"] if not is_junk(x)])
    player["competitions"] = unique([x for x in player["competitions"] if not is_junk(x)])

    # Een antwoord moet bij twee aanwijzingen kunnen passen.
    relevant_values = []
    for field in ("clubs","internationalTeams","awards","competitions"):
        relevant_values.extend(v for v in player[field] if norm(v) in allowed_norm)

    if len({norm(v) for v in relevant_values}) < 2:
        continue

    # Verwijder niet-speelbare clue-waarden uit de runtime database.
    for field in ("clubs","internationalTeams","awards","competitions"):
        player[field] = [v for v in player[field] if norm(v) in allowed_norm]

    players.append(player)

players.sort(key=lambda p: norm(p["name"]))

# Runtime indexes vooraf bouwen, zodat de browser dit niet meer hoeft te doen.
clue_players = {}
name_lookup = {}

    names = [player["name"], *player["aliases"]]
    for name in names:
        key = norm(clean_name(name))
        if key:
            name_lookup.setdefault(key, []).append(index)

    for field in ("clubs","internationalTeams","awards","competitions"):
        for clue in player[field]:
            key = norm(clue)
            clue_players.setdefault(key, []).append(index)
